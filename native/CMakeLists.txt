cmake_minimum_required(VERSION 3.22)
project(VolleyScopeNative VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VS_BUILD_APP "Build the native Qt/ONNX player" ON)
add_library(volleyscope_core core/tracking.cpp)
target_include_directories(volleyscope_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
 target_compile_options(volleyscope_core PUBLIC /utf-8 /W4)
else()
 target_compile_options(volleyscope_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE volleyscope_core)
add_test(NAME identity_and_ball_regressions COMMAND core_tests)
if(VS_BUILD_APP)
 set(CMAKE_AUTOMOC ON)
 find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Widgets Network Multimedia Concurrent Sql)
 set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime package directory")
 find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include ${ONNXRUNTIME_ROOT}/build/native/include REQUIRED NO_DEFAULT_PATH)
 find_library(ORT_LIBRARY NAMES onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib ${ONNXRUNTIME_ROOT}/runtimes/win-x64/native REQUIRED NO_DEFAULT_PATH)
 add_executable(VolleyScope WIN32 app/main.cpp app/window.cpp app/project.cpp app/models.cpp app/worker.cpp app/calibration.cpp)
 target_include_directories(VolleyScope PRIVATE ${ORT_INCLUDE})
 target_link_libraries(VolleyScope PRIVATE volleyscope_core Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Multimedia Qt6::Concurrent Qt6::Sql ${ORT_LIBRARY})
 if(WIN32)
  target_compile_definitions(VolleyScope PRIVATE VS_DIRECTML=1 NOMINMAX WIN32_LEAN_AND_MEAN)
 endif()
 if(MSVC)
  target_compile_options(VolleyScope PRIVATE /utf-8 /W4)
 endif()
 add_test(NAME native_model_and_media_checks COMMAND VolleyScope --self-test --models ${CMAKE_CURRENT_SOURCE_DIR}/models)
 set_tests_properties(native_model_and_media_checks PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
